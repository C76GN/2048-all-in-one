extends RefCounted


func mixes_allowed_and_wrong_build_probe() -> bool:
	return OS.has_feature("with_dev_tools") or OS.has_feature("web")


func compares_display_name_for_wrong_platform() -> bool:
	return DisplayServer.get_name() == "windows"


func exposes_raw_display_name() -> String:
	return DisplayServer.get_name()
